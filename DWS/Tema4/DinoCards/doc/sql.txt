-- SCRIPT de Generacion de BBDD en MySQL
CREATE DATABASE IF NOT EXISTS DinoCards;

-- DROPS
DROP TABLE IF EXISTS colecciones;
DROP TABLE IF EXISTS dinosaurios;
DROP TABLE IF EXISTS usuarios;

-- CREATE TABLES
CREATE TABLE usuarios(id INT primary key auto_increment,
					  username VARCHAR(64)not null UNIQUE,
                      email VARCHAR(256)not null UNIQUE,
                      password_hash VARCHAR(1024) not null,
                      fecha_registro DATETIME,
                      ultima_obtencion DATETIME);
CREATE TABLE dinosaurios(id INT primary key auto_increment,
					  nombre VARCHAR(64),
                      especie VARCHAR(1024),
                      periodo VARCHAR(1024),
                      image_url VARCHAR(1024),
                      altura DECIMAL(4,2),
                      largo DECIMAL(4,2),
                      peso DECIMAL(6,3),
                      hp INT,
                      vigor INT,
                      ataque INT,
                      defensa INT,
                      agilidad INT);
CREATE TABLE colecciones(id INT primary key auto_increment,
						usuario_id INT,
						dinosaurio_id INT,
                        fecha_adquisicion DATETIME,
              foreign key (usuario_id) REFERENCES usuarios(id),
              foreign key (dinosaurio_id) REFERENCES dinosaurios(id)
			);


-- PROCEDIMIENTOS ALMACENADOS
	-- PA REGISTRO
DROP PROCEDURE IF EXISTS Registro;
DELIMITER $$
CREATE PROCEDURE Registro(IN _username VARCHAR(64),
                          IN _email VARCHAR(256),
                          IN _password VARCHAR(1024),
                          OUT _res INT
                          )
BEGIN
	DECLARE aux INT;

    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        SET _res = -99;
    END;

    SET aux = 0;

    IF (_username = "" OR _email = "") THEN
        SET _res = -1;
    ELSEIF (_password = "") THEN
        SET _res = -1;
    ELSE
        SELECT COUNT(*) INTO aux FROM usuarios 
        WHERE username = _username OR email = _email;

        IF aux > 0 THEN
            SET _res = -2;
        ELSE
            INSERT INTO usuarios(username, email, password_hash, fecha_registro)
            VALUES(_username, _email, _password, SYSDATE());
            SET _res = 0;
        END IF;
    END IF;
END$$
DELIMITER ;

    -- PA LOGIN
DROP PROCEDURE IF EXISTS Login;
DELIMITER $$
CREATE PROCEDURE Login(IN _email VARCHAR(256),
						IN _password VARCHAR(1024),
						OUT _res INT
                        )
BEGIN
	DECLARE aux VARCHAR(64);

    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        SET _res = -99;
    END;

    SET aux = NULL;

    SELECT username FROM usuarios WHERE email = _email and password_hash = _password
    INTO aux;

    IF(_email = "") THEN
		SET _res = -1;
	ELSEIF(_password = "") THEN
		SET _res = -1;
    ELSEIF(aux IS NULL) THEN
		SET _res = -3;
	ELSE
		SET _res = 0;
        SELECT * FROM usuarios WHERE email = _email and password_hash = _password;
	end if;
END$$
DELIMITER ;

    -- PA AñadirNuevaCartaAColeccion
DROP PROCEDURE IF EXISTS AnadirNuevaCartaAColeccion;
DELIMITER $$
CREATE PROCEDURE AnadirNuevaCartaAColeccion(IN _usuario_id INT,
											OUT _res INT
											)
BEGIN
DECLARE auxUsuarioExiste INT;
DECLARE v_ultima_obtencion DATETIME;

DECLARE EXIT HANDLER FOR SQLEXCEPTION
BEGIN
    SET _res = -99;
END;

SET auxUsuarioExiste = NULL;
SET v_ultima_obtencion = NULL;

SELECT id, ultima_obtencion INTO auxUsuarioExiste, v_ultima_obtencion
FROM usuarios WHERE id = _usuario_id;

    IF(auxUsuarioExiste IS NULL) THEN
		SET _res = -8;
    ELSE
		if(TIMESTAMPDIFF(SECOND, v_ultima_obtencion, NOW()) >= 24 or v_ultima_obtencion is null)
        then
		INSERT INTO colecciones(usuario_id,dinosaurio_id,fecha_adquisicion)
						VALUES(_usuario_id, floor(1+rand()*100),SYSDATE());
		update usuarios set ultima_obtencion=sysdate()
		where id=_usuario_id;
		SET _res = 0;
        else set _res=-9;
		end if;
    END IF;
END$$
DELIMITER ;

    -- PA Obtener todas las cartas con estado
DROP PROCEDURE IF EXISTS ObtenerTodasLasCartasConEstado;
DELIMITER $$
CREATE PROCEDURE ObtenerTodasLasCartasConEstado(IN _usuario_id INT, OUT _res INT)
BEGIN
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        SET _res = -99;
    END;

    SET _res = 0;

    SELECT d.*,
           CASE WHEN c.usuario_id IS NOT NULL THEN 1 ELSE 0 END AS tiene
    FROM dinosaurios d
    LEFT JOIN colecciones c
        ON d.id = c.dinosaurio_id AND c.usuario_id = _usuario_id
    ORDER BY d.id;
END$$
DELIMITER ;

    -- PA ObtenerColeccion
DROP PROCEDURE IF EXISTS ObtenerColeccion;
DELIMITER $$
CREATE PROCEDURE ObtenerColeccion(IN _usuario_id INT,
								  OUT _res INT
								  )
BEGIN
	DECLARE aux int;

    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        SET _res = -99;
    END;

    SET aux = NULL;

    select id from usuarios where id=_usuario_id
    into aux;

    if(aux is null)
    then 
    set _res=-10;
		else
        SET _res = 0;
        SELECT d.* FROM colecciones c
		JOIN dinosaurios d ON c.dinosaurio_id = d.id
		WHERE c.usuario_id = _usuario_id;
    end if;    
END$$
DELIMITER ;
